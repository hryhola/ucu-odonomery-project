\section{Linear-algebra formulation}
\label{sec:formulation}

\guide{Criterion A. State the problem precisely as a linear-algebra
problem: the matrix (its size, its entries, its structure: sparse,
symmetric, positive definite, low rank), the unknowns, and what you need
to do with the matrix. Explain why this structure suggests your methods.
\emph{Interim report 1}: this section is the core of the memo.}

Number every displayed formula you refer to, and refer to it by number,
as in~\eqref{eq:ls} below. Use \verb|\begin{equation}| or
\verb|\begin{align}|, never \verb|$$ ... $$|. For example, the least
squares problem with a matrix $A\in\bR^{m\times n}$ of full column rank
and its solution via the thin QR factorization $A=QR$ read
\begin{equation}\label{eq:ls}
  \min_{\bx\in\bR^n}\|A\bx-\bb\|_2,
  \qquad
  R\bx = Q^{\top}\bb .
\end{equation}

\begin{definition}\label{def:cond}
The condition number of a matrix $A$ of full column rank is
$\cond(A)=\sigma_{\max}(A)/\sigma_{\min}(A)$.
\end{definition}

\begin{remark}
Since $\cond(A^{\top}A)=\cond(A)^2$, solving~\eqref{eq:ls} through the
normal equations can lose twice as many digits as solving it through QR.
\end{remark}

\subsection{Algorithm 1}

Present an algorithm either as pseudocode, as in
Algorithm~\ref{alg:power}, or as a short code listing, as in
Listing~\ref{lst:power}. Annotate the expensive steps with their cost.

\begin{algorithm}[ht]
  \caption{Power iteration for the dominant eigenpair}\label{alg:power}
  \KwIn{$A\in\bR^{n\times n}$, starting vector $\bx_0\neq 0$,
        tolerance $\varepsilon>0$}
  \KwOut{approximate dominant eigenpair $(\lambda,\bx)$}
  $\bx\gets\bx_0/\|\bx_0\|$;\quad $\by\gets A\bx$;\quad
  $\lambda\gets\bx^{\top}\by$\;
  \While{$\|\by-\lambda\bx\|>\varepsilon\,|\lambda|$}{
    $\bx\gets\by/\|\by\|$\;
    $\by\gets A\bx$\tcp*{one matrix-vector product: $\sim 2\nnz(A)$ flops}
    $\lambda\gets\bx^{\top}\by$\tcp*{Rayleigh quotient: $2n$ flops}
  }
  \Return{$\lambda$, $\bx$}
\end{algorithm}

\medskip\noindent\textbf{Cost.} About $2\nnz(A)$ flops per iteration and $O(n)$
memory besides $A$. The error decreases like $|\lambda_2/\lambda_1|^k$,
so the iteration is slow when the two largest eigenvalues are close in
absolute value.

\begin{lstlisting}[caption={Power iteration in NumPy.},label={lst:power}]
import numpy as np

def power_iteration(A, x0, tol=1e-10, maxit=1000):
    x = x0 / np.linalg.norm(x0)
    y = A @ x                     # the only expensive step
    lam = x @ y
    for _ in range(maxit):
        if np.linalg.norm(y - lam * x) <= tol * abs(lam):
            break
        x = y / np.linalg.norm(y)
        y = A @ x
        lam = x @ y
    return lam, x
\end{lstlisting}
